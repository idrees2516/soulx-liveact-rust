    68→def resample_audio(audio, sr, fps):
    69→    rate = 25 / fps
    70→    y, sr_out = torchaudio.sox_effects.apply_effects_tensor(audio, sr, [["tempo", f"{rate}"]])
    71→    resampler = T.Resample(sr_out, 16000).to(audio.device)
    72→    return resampler(y) * 3.0, 16000

   212→    def _warmup(self):
   213→        print(f"[Warmup][Rank {self.rank}] start", flush=True)
   214→
   215→        if dist.is_initialized():
   216→            dist.barrier()
   217→        torch.cuda.empty_cache()
   218→        torch.cuda.synchronize(self.device)
   219→
   220→        try:
   221→            with torch.no_grad():
   222→                frame_num_init = (sum(self.blksz_lst) - 1) * 4 + 1
   223→                # 1. 准备假图像
   224→                cond_image = torch.randn(
   225→                    1, 3, 1, self.height, self.width,
   226→                    device=self.device, dtype=torch.bfloat16
   227→                ).clamp_(-1, 1)
   228→                # 2. CLIP
   229→                with torch.autocast("cuda", dtype=torch.bfloat16):
   230→                    clip_context = self.clip.visual(cond_image)
   231→
   232→                # 3. 假音频
   233→                dummy_audio = torch.randn(16000 * 6)
   234→                audio_embedding = get_embedding(
   235→                    dummy_audio,
   236→                    self.wav2vec_feature_extractor,
   237→                    self.audio_encoder,
   238→                    device=self.device)
   239→                # 4. init y
   240→                ref_target_masks = torch.ones(
   241→                    3,
   242→                    self.height // self.vae_stride[1],
   243→                    self.width // self.vae_stride[2],
   244→                    device=self.device,
   245→                    dtype=torch.bfloat16)
   246→
   247→                video_frames_placeholder = torch.zeros(
   248→                    1,
   249→                    cond_image.shape[1],
   250→                    frame_num_init - cond_image.shape[2],
   251→                    self.height,
   252→                    self.width,
   253→                    device=self.device,
   254→                    dtype=torch.bfloat16)
   255→
   256→                padding_frames = torch.concat([cond_image, video_frames_placeholder], dim=2)
   257→
   258→                with torch.autocast("cuda", dtype=torch.bfloat16):
   259→                    y = self.vae.encode(padding_frames).to(self.device).unsqueeze(0)
   260→
   261→                msk = get_msk(frame_num_init, cond_image, self.vae_stride, self.device)
   262→                y = torch.concat([msk, y], dim=1)
   263→
   264→                # 5. prompt
   265→                context = [
   266→                    self.text_encoder(
   267→                        texts="A person is speaking naturally.",
   268→                        device='cpu' if self.args.t5_cpu else self.device
   269→                    )[0].to(self.device, dtype=torch.bfloat16)
   270→                ]
   271→
   272→                # 6. 完全按原逻辑跑，只是 iter_total_num = 2
   273→                iter_total_num = 2
   274→                pre_latent = None
   275→
   276→                for iteration in range(iter_total_num):
   277→                    audio_start_idx = 0 if iteration == 0 else (iteration - 1) * self.blksz_lst[-1] * self.vae_stride[0]
   278→                    audio_end_idx = audio_start_idx + frame_num_init
   279→
   280→                    audio_embs = get_audio_emb(audio_embedding, audio_start_idx, audio_end_idx, self.device)
   281→
   282→                    y_cut = y[:, :, :frame_num_init // 4 + 1, ...]
   283→                    f_idx = 0 if iteration == 0 else 1
   284→                    latent = torch.randn(
   285→                        16,
   286→                        self.blksz_lst[f_idx],
   287→                        self.height // self.vae_stride[1],
   288→                        self.width // self.vae_stride[2],
   289→                        dtype=torch.bfloat16,
   290→                        device=self.device
   291→                    )
   292→
   293→                    with torch.autocast("cuda", dtype=torch.bfloat16):
   294→                        for i in range(len(self.timesteps) - 1):
   295→                            timestep = self.timesteps[i]
   296→                            arg_c = {
   297→                                'context': context,
   298→                                'clip_fea': clip_context,
   299→                                'ref_target_masks': ref_target_masks,
   300→                                'audio': audio_embs,
   301→                                'y': y_cut[:, :, sum(self.blksz_lst[:f_idx]):sum(self.blksz_lst[:f_idx + 1])],
   302→                                'start_idx': sum(self.blksz_lst[:f_idx]) * self.frame_len,
   303→                                'end_idx': sum(self.blksz_lst[:f_idx + 1]) * self.frame_len,
   304→                                'update_cache': iteration > 1
   305→                            }
   306→
   307→                            noise_pred = self.wan_i2v_model(
   308→                                [latent],
   309→                                t=timestep,
   310→                                kv_cache=self.kv_cache[i],
   311→                                skip_audio=False if i in [1, 2] else True,
   312→                                **arg_c
   313→                            )[0]
   314→
   315→                            dt = (self.timesteps[i] - self.timesteps[i + 1]) / 1000
   316→                            latent = latent + (-noise_pred) * dt[0]
   317→
   318→                        if iteration == 0:
   319→                            _videos = self.vae.decode(latent)
   320→                        else:
   321→                            combined_latent = torch.concat([pre_latent[:, -3:], latent], dim=1)
   322→                            _videos = self.vae.decode(combined_latent)[:, :, 9:]
   323→
   324→                        pre_latent = latent
   325→                    torch.cuda.synchronize(self.device)
   326→                    print(f"[Warmup][Rank {self.rank}] iteration {iteration + 1}/2 done", flush=True)
   327→
   328→                del cond_image, clip_context, dummy_audio, audio_embedding
   329→                del ref_target_masks, video_frames_placeholder, padding_frames
   330→                del y, msk, context, audio_embs, y_cut, latent, pre_latent, _videos
   331→                if 'combined_latent' in locals():
   332→                    del combined_latent
   333→                if 'noise_pred' in locals():
   334→                    del noise_pred
   335→
   336→                # torch.cuda.empty_cache()
   337→                torch.cuda.synchronize(self.device)
   338→
   339→            if dist.is_initialized():
   340→                dist.barrier()
   341→
   342→            print(f"[Warmup][Rank {self.rank}] done", flush=True)
   343→        except Exception as e:
   344→            print(f"[Warmup][Rank {self.rank}] failed: {e}", flush=True)
   345→            raise

   365→        def close_proc(proc, name="ffmpeg"):
   366→            if proc is None:
   367→                return
   368→            try:
   369→                if proc.stdin:
   370→                    proc.stdin.close()
   371→            except Exception:
   372→                pass
   373→            try:
   374→                ret = proc.wait()
   375→                if ret != 0:
   376→                    print(f"[{name}] exited with code {ret}", flush=True)
   377→            except Exception as e:
   378→                print(f"[{name}] wait failed: {e}", flush=True)
   379→
   380→        def write_chunk_bytes(proc, chunk_bytes, name="ffmpeg"):
   381→            if proc is None or proc.stdin is None:
   382→                return
   383→            try:
   384→                proc.stdin.write(chunk_bytes)
   385→                proc.stdin.flush()
   386→            except BrokenPipeError:
   387→                raise RuntimeError(f"{name} stdin broken pipe")
   388→            except Exception as e:
   389→                raise RuntimeError(f"write to {name} failed: {e}")

   391→        def tensor_chunk_to_rgb_bytes(video_tensor):
   392→            """
   393→            video_tensor: [1, 3, T, H, W] in [-1, 1]
   394→            return:
   395→                chunk_bytes: 整个 chunk 的连续 rgb24 bytes
   396→                num_frames: 这个 chunk 的帧数
   397→            """
   398→            video_u8 = (
   399→                ((video_tensor.squeeze(0).permute(1, 2, 3, 0) + 1.0) * 127.5)
   400→                .clamp(0, 255)
   401→                .to(torch.uint8)
   402→                .contiguous()
   403→                .cpu()
   404→            )  # [T, H, W, C], uint8
   405→            num_frames = video_u8.shape[0]
   406→            chunk_bytes = video_u8.numpy().tobytes()
   407→            return chunk_bytes, num_frames
